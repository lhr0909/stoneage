cmake_minimum_required(VERSION 3.20)
project(StoneAge25Native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CLIENT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../石器时代8.5客户端最新源代码/石器源码")
add_library(stoneage25-codec STATIC "${CLIENT_SOURCE}/newproto/autil.cpp")
target_compile_definitions(stoneage25-codec PUBLIC _SA_VERSION_25 _SA_VERSION_SPECIAL STONEAGE_LOCAL_25 STONEAGE_PROTOCOL_ONLY)
target_include_directories(stoneage25-codec PUBLIC "${CLIENT_SOURCE}/newproto" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(stoneage25-codec PRIVATE -Wno-writable-strings -Wno-deprecated-declarations)
add_executable(stoneage25-probe protocol_probe.cpp)
target_link_libraries(stoneage25-probe PRIVATE stoneage25-codec)

option(STONEAGE_SANITIZERS "Enable address/undefined behavior checks" OFF)
if(STONEAGE_SANITIZERS)
  target_compile_options(stoneage25-codec PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(stoneage25-probe PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(stoneage25-probe PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME codec COMMAND stoneage25-probe --self-test)
